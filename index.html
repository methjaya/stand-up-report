/* Meeting Minutes PDF Generator (Fixed Overflow & Page Breaks) */ function generateMOMPDF() { const { jsPDF } = window.jspdf; const doc = new jsPDF(); const title = activePdfData ? activePdfData.title : (document.getElementById('momTitle').value || "Meeting Minutes"); const rawAttendees = activePdfData ? activePdfData.attendees : (document.getElementById('momAttendees').value || "All Members"); const date = activePdfData ? activePdfData.date : (document.getElementById('momDate').value || new Date().toISOString().split('T')[0]); const time = activePdfData ? activePdfData.time : (document.getElementById('momTime').value || "N/A"); const discussions = activePdfData ? activePdfData.discussions : (document.getElementById('momDiscussions').value || "None recorded."); const actions = activePdfData ? activePdfData.actions : (document.getElementById('momActions').value || "None recorded."); let y = 20; // Header Banner doc.setFillColor(28, 28, 30); doc.rect(0, 0, 210, 42, "F"); doc.setFontSize(16); doc.setTextColor(255, 255, 255); doc.setFont("helvetica", "bold"); doc.text("MEETING MINUTES (MOM)", 16, y); doc.setFontSize(11); doc.setTextColor(142, 142, 147); doc.setFont("helvetica", "normal"); doc.text("Sahansa The PM's Management Suite", 16, y + 7); y = 52; // --- ATTENDEES & METADATA BLOCK --- const parties = rawAttendees.split('|').map(p => p.trim()).filter(Boolean); let wrappedPartiesText = []; parties.forEach(p => { const splitP = doc.splitTextToSize(p, 130); wrappedPartiesText = wrappedPartiesText.concat(splitP); }); const attendeesBoxHeight = 18 + (wrappedPartiesText.length * 4.5); doc.setFillColor(242, 242, 247); doc.roundedRect(16, y, 178, attendeesBoxHeight, 2, 2, "F"); doc.setFontSize(9); doc.setTextColor(108, 108, 112); doc.setFont("helvetica", "bold"); doc.text("TITLE:", 22, y + 8); doc.setFont("helvetica", "normal"); doc.setTextColor(28, 28, 30); doc.text(doc.splitTextToSize(title, 80), 36, y + 8); doc.setFont("helvetica", "bold"); doc.setTextColor(108, 108, 112); doc.text("DATE & TIME:", 120, y + 8); doc.setFont("helvetica", "normal"); doc.setTextColor(28, 28, 30); doc.text(`${date} @ ${time}`, 145, y + 8); doc.setFont("helvetica", "bold"); doc.setTextColor(108, 108, 112); doc.text("ATTENDEES:", 22, y + 16); doc.setFont("helvetica", "normal"); doc.setTextColor(28, 28, 30); let partyY = y + 16; wrappedPartiesText.forEach((line, idx) => { doc.text(line, 46, partyY + (idx * 4.5)); }); y += attendeesBoxHeight + 10; // Helper to render long text sections with automatic boxes & multipage handling function renderSection(sectionTitle, textContent) { doc.setFontSize(11); doc.setFont("helvetica", "bold"); doc.setTextColor(0, 113, 227); // Page break check for section title if (y + 15 > 275) { doc.addPage(); y = 20; } doc.text(sectionTitle, 16, y); y += 6; doc.setFontSize(8.5); doc.setFont("helvetica", "normal"); doc.setTextColor(28, 28, 30); const splitLines = doc.splitTextToSize(textContent, 168); const lineHeight = 4.5; const totalTextHeight = splitLines.length * lineHeight; const boxHeight = totalTextHeight + 8; // If the box fits completely on the current page if (y + boxHeight <= 275) { doc.setFillColor(242, 242, 247); doc.setDrawColor(229, 229, 234); doc.roundedRect(16, y, 178, boxHeight, 2, 2, "FD"); doc.text(splitLines, 20, y + 6); y += boxHeight + 10; } else { // Split multiline content across pages let lineIndex = 0; while (lineIndex < splitLines.length) { const spaceRemaining = 275 - y; const linesThatFit = Math.floor((spaceRemaining - 8) / lineHeight); if (linesThatFit <= 0) { doc.addPage(); y = 20; continue; } const chunk = splitLines.slice(lineIndex, lineIndex + linesThatFit); const chunkBoxHeight = (chunk.length * lineHeight) + 8; doc.setFillColor(242, 242, 247); doc.setDrawColor(229, 229, 234); doc.roundedRect(16, y, 178, chunkBoxHeight, 2, 2, "FD"); doc.text(chunk, 20, y + 6); lineIndex += linesThatFit; y += chunkBoxHeight + 6; if (lineIndex < splitLines.length) { doc.addPage(); y = 20; } } y += 4; } } // --- KEY DISCUSSIONS & DECISIONS --- renderSection("KEY DISCUSSIONS & DECISIONS", discussions); // --- ACTION ITEMS & RESPONSIBILITIES --- renderSection("ACTION ITEMS & RESPONSIBILITIES", actions); doc.save(`Meeting_Minutes_${date}.pdf`); activePdfData = null; }
